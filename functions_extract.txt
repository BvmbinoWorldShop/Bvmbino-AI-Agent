function toggleLeftSidebar(val) {
  const sb = document.getElementById('sb');
  if (val !== undefined) {
    if (val) sb.classList.add('collapsed'); else sb.classList.remove('collapsed');
  } else {
    sb.classList.toggle('collapsed');
  }
  localStorage.setItem('bvmbino_sb_left_collapsed', sb.classList.contains('collapsed'));
}

}

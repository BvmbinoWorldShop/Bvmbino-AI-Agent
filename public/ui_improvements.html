// In chat.js backend — add streaming support
export default async function handler(req, res) {
  // ... existing CORS headers ...

  // Set streaming headers
  res.setHeader('Content-Type', 'text/event-stream');
  res.setHeader('Cache-Control', 'no-cache');
  res.setHeader('Connection', 'keep-alive');

  // Groq supports streaming
  const r = await fetch('https://api.groq.com/openai/v1/chat/completions', {
    method: 'POST',
    headers: { Authorization: `Bearer ${groqKey}`, 'Content-Type': 'application/json' },
    body: JSON.stringify({ model: 'llama-3.3-70b-versatile', messages, stream: true, max_tokens: 8000 })
  });

  // Pipe SSE stream directly to client
  for await (const chunk of r.body) {
    res.write(chunk);
  }
  res.end();
}

// In index.html — replace appendAiText with streaming reader
async function callAIStream() {
  const response = await fetch('/api/chat', { method: 'POST', body: JSON.stringify({ messages }) });
  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let aiText = '';

  // Create the bubble first (empty)
  const bubble = appendEmptyAiBubble();

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    const lines = decoder.decode(value).split('\n').filter(l => l.startsWith('data: '));
    for (const line of lines) {
      const data = line.slice(6);
      if (data === '[DONE]') continue;
      const delta = JSON.parse(data).choices?.[0]?.delta?.content || '';
      aiText += delta;
      bubble.innerHTML = renderMarkdown(aiText) + '<span class="cursor"></span>';
      scrollBot();
    }
  }
  bubble.innerHTML = renderMarkdown(aiText); // Remove cursor when done
}

let recognition = null;
function toggleVoice() {
  const btn = document.getElementById('micbtn');
  if (!recognition) {
    if (!('webkitSpeechRecognition' in window || 'SpeechRecognition' in window)) {
      toast('Voice not supported in this browser'); return;
    }
    recognition = new (window.SpeechRecognition || window.webkitSpeechRecognition)();
    recognition.lang = 'fr-FR'; // or 'en-US' — auto-detect from page language
    recognition.interimResults = true;
    recognition.onresult = e => {
      const transcript = [...e.results].map(r => r[0].transcript).join('');
      document.getElementById('inp').value = transcript;
      onInp(); // trigger auto-resize
    };
    recognition.onend = () => {
      btn.textContent = '🎙️'; btn.style.color = '';
      // Auto-send when voice stops (optional — remove if prefer manual send)
      if (document.getElementById('inp').value.trim()) send();
    };
  }
  if (btn.textContent === '⏹️') {
    recognition.stop(); return;
  }
  btn.textContent = '⏹️'; btn.style.color = '#ef4444';
  recognition.start();
}

const STORAGE_KEY = 'bvmbino_sessions';
const MAX_SESSIONS = 20;

function saveSession(title, messages) {
  const sessions = JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');
  const session = { id: Date.now(), title: title.slice(0, 60), messages, date: new Date().toISOString() };
  sessions.unshift(session); // newest first
  if (sessions.length > MAX_SESSIONS) sessions.pop(); // keep last 20
  localStorage.setItem(STORAGE_KEY, JSON.stringify(sessions));
  renderHistory();
}

function renderHistory() {
  const sessions = JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');
  const hist = document.getElementById('hist');
  hist.innerHTML = sessions.map(s =>
    `<div class="hi" onclick="loadSession(${s.id})">
      <span class="hdot"></span>
      <span style="flex:1;overflow:hidden;text-overflow:ellipsis">${esc(s.title)}</span>
      <button onclick="event.stopPropagation();deleteSession(${s.id})"
        style="opacity:0;font-size:10px;background:none;border:none;color:var(--t3);cursor:pointer"
        onmouseenter="this.style.opacity=1" onmouseleave="this.style.opacity=0">✕</button>
    </div>`
  ).join('');
}

function loadSession(id) {
  const sessions = JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');
  const session = sessions.find(s => s.id === id);
  if (!session) return;
  msgs = session.messages;
  // Re-render all messages
  document.getElementById('ci').innerHTML = '';
  msgs.forEach(m => m.role === 'user' ? addUserMsg(m.content) : appendAiText(m.content, null));
  document.getElementById('tbtitle').textContent = session.title;
  closeSb();
}

🟢 POLISH 6. Swipe Gestures on Mobile
👆 Swipe Right to Open Sidebar
On mobile, the sidebar is hidden behind the ☰ hamburger button. Adding swipe gestures (swipe right → open sidebar, swipe left → close) makes navigation feel native. This is the standard mobile pattern users expect from modern apps.
🔥 Native mobile feel 30 min · No libraries needed
Add at the bottom of the script — works on any touch device
// ── SWIPE GESTURES (mobile) ──
let touchStartX = 0;
let touchStartY = 0;

document.addEventListener('touchstart', e => {
  touchStartX = e.touches[0].clientX;
  touchStartY = e.touches[0].clientY;
}, { passive: true });

document.addEventListener('touchend', e => {
  const dx = e.changedTouches[0].clientX - touchStartX;
  const dy = e.changedTouches[0].clientY - touchStartY;

  // Only register horizontal swipes (dx > 2× vertical movement)
  if (Math.abs(dx) < Math.abs(dy) * 2) return;
  if (Math.abs(dx) < 50) return; // minimum swipe distance

  if (dx > 0 && touchStartX < 40) {
    // ← Swipe RIGHT from left edge → open sidebar
    openSb();
  } else if (dx < 0) {
    // ← Swipe LEFT anywhere → close sidebar
    closeSb();
  }
}, { passive: true });
